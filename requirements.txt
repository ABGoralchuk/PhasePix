# Streamlit dashboard
#
# Pinned to 1.39.0 ON PURPOSE. Streamlit 1.39 is the last release where
# `image_to_url` still lives in `streamlit.elements.image` with the legacy
# positional signature -- exactly what `streamlit-drawable-canvas==0.9.3` calls
# for its background image. From 1.40 onwards that helper moved to
# `streamlit.elements.lib.image_utils` with a different signature, which is what
# produced the blank / white canvas in the Manual Correction tab. No monkey
# patching is used anywhere in this project.
#
# Trade-off accepted by the maintainers: this is a local, single-user scientific
# desktop tool bound to localhost, so the Streamlit advisories fixed in 1.54.0
# (CVE-2026-33682 Windows SSRF / NTLM relay) do not apply to its deployment
# model. Never expose this container to an untrusted network.
streamlit==1.39.0

# Fabric.js drawing canvas. Natively compatible with the pinned Streamlit 1.39.
streamlit-drawable-canvas==0.9.3

# ---------------------------------------------------------------------------
# Deep learning segmentation (Cellpose + Segment Anything)
# ---------------------------------------------------------------------------
# torch / torchvision are intentionally LEFT UNPINNED (no version constraints).
#
# The Windows portable builder (build_portable_windows.bat) installs the CUDA
# 12.1 wheels BEFORE this file is processed:
#     pip install torch torchvision --index-url https://download.pytorch.org/whl/cu121
# An unconstrained requirement is already satisfied by the installed build
# (e.g. 2.5.1+cu121), so `pip install -r requirements.txt` keeps the CUDA
# wheels instead of pulling the CPU-only wheels from PyPI. Never add a bound
# here (e.g. `torch>=2.6`): any constraint the +cu121 build cannot satisfy makes
# pip swap in CPU wheels and GPU acceleration silently disappears. The builder
# re-verifies torch.version.cuda after this step and repairs it if needed.
torch
torchvision
cellpose
segment-anything
